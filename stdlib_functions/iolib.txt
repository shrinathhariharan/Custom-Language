import io

// writes to a file (creates or overwrites)
myFile = io.write("output.txt")
myFile.write("Hello World\n") // writes text to file
myFile.writeLine("Second line of text") // writes line with newline
myFile.flush() // flushes stream buffer
myFile.close() // closes file stream

// reads from a file
inFile = io.read("output.txt")
str content = inFile.read() // reads entire file content
inFile.close()

// reading line-by-line
inFile2 = io.read("output.txt")
str line1 = inFile2.readLine() // reads single line
str line2 = inFile2.readLine()
inFile2.close()

// reading all lines into a dynamic array
inFile3 = io.read("output.txt")
lines = inFile3.readLines() // reads all lines into dynamic array
inFile3.close()

// appending to a file
appFile = io.append("output.txt")
appFile.write("Appended content\n")
appFile.close()

// file checking and removal
bool exists = io.exists("output.txt") // checks if file exists on disk
bool removed = io.remove("output.txt") // deletes file from disk
