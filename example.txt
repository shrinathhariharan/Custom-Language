// Variable declarations
int x = 10
dec y = 3.14
str z = "hello"
bool flag = true

print(x + "\n")
print(y + "\n")
print(z + "\n")
print(flag + "\n")

// Dynamic arrays
int nums[] = {1, 2, 3, 4, 5}
print(nums + "\n")
print(nums[2] + "\n")

// Compound assignment (also works on array items)
x += 5
y *= 2
nums[0] += 10
print("x after +=: " + x + "\n")
print("y after *=: " + y + "\n")
print("first number after +=: " + nums[0] + "\n")

// Array member functions
print("array size: " + nums.size() + "\n")
nums.push(6)
nums.insert(1, 99) // insert takes (index, item)
print("after push and insert: " + nums + "\n")
nums.remove(99) // removes first occurrence of 99
print("after removing 99: " + nums + "\n")
int popped = nums.pop()
print("popped item: " + popped + "\n")
print("array size after pop: " + nums.size() + "\n")

str words[] = {"apple", "banana", "cherry"}
print(words[0] + "\n")

// Logical operators and else-if blocks
if (flag && nums.size() > 0) {
    print("The array has items and flag is true\n")
} else if (!flag || nums.size() == 0) {
    print("Flag is false or the array is empty\n")
} else {
    print("This branch will not run for the current values\n")
}

// Classes have public and private access methods, and member functions
class Point {
    int _x = 0
    int _y = 0
    int example = 0

    Point(self, x, y) {
        self._x = x
        self._y = y
    }

    func void hello() {
        print("Hello from Point\n")
    }
}

Point point = Point(3, 4)
point.example += 1
point.hello()
print("public field: " + point.example + "\n")

// Interactive input

// Functions with return values
func int add(int a, int b) 
{
    return a + b
}

func str greet(str name) 
{
    return "Hello, " + name
}

int result = add(3, 7)
print(result + "\n")
print(greet("World") + "\n")

// Loops
int count = 0
while (count < 3) {
    print("While loop count: ")
    print(count + "\n")
    count += 1
}

for (int i = 0, i < 4, i += 1) {
    print("For loop count: ")
    print(i + "\n")
}

// continue skips the rest of this iteration; break exits the loop
int control = 0
while (control < 6) {
    control += 1
    if (control == 2) {
        continue
    }
    if (control == 5) {
        break
    }
    print("Loop control: " + control + "\n")
}

// Void-style function (no return value)
func void printLine(str msg) 
{
    print(msg)
}

printLine("Done!\n")

// Type casting and type inspection
str numStr = "99"
int asInt = toInt(numStr)       // "99" -> 99
dec asDec = toDec(numStr)       // "99" -> 99.0
bool asBool = toBool(numStr)    // non-empty string -> true
str backToStr = toStr(asInt)    // 99 -> "99"

print("toInt(\"99\"):  " + asInt + "\n")
print("toDec(\"99\"):  " + asDec + "\n")
print("toBool(\"99\"): " + asBool + "\n")
print("toStr(99):    " + backToStr + "\n")

// toBool rules: 0 and "" are false, everything else is true
print("toBool(0):    " + toBool(0) + "\n")
print("toBool(\"\"):   " + toBool("") + "\n")
print("toBool(3.14): " + toBool(3.14) + "\n")

// type() returns the runtime type name as a string
int  ti = 1

print("type(ts): " + type(ts) + "\n")
print("type(tb): " + type(tb) + "\n")
